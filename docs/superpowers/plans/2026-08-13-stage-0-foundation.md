# Stage 0 — Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up a compiling, tested, locally-running Go backend skeleton whose HTTP surface is generated from the frontend's `openapi.json`, with Postgres + ClickHouse + Redis wired, RLS tenant isolation proven by a failing-to-leak test, and a green `make check`.

**Architecture:** Two Go binaries (`control-api`, `worker`) plus a shared `internal/` library. The HTTP layer is generated by `oapi-codegen` from the frontend's spec — implemented operations are compile-checked against it; unimplemented ones return 501 from a generated base struct. Postgres holds the control plane behind row-level security; ClickHouse holds message logs; Redis holds hot state. No Docker locally — native brew services.

**Tech Stack:** Go 1.23+ · chi v5 · oapi-codegen v2 · sqlc · goose · pgx v5 · River · PostgreSQL 16 · ClickHouse · Redis 8 · testify

**Spec:** `docs/BACKEND_DESIGN.md`, `docs/ARCHITECTURE.md`, `docs/RESEARCH_AND_SCALE.md`; contract is `../SMS-UI/openapi.json`

## Global Constraints

- Go module path: `github.com/saeedafri/sms-be`
- Go version floor: `1.23`
- **`internal/gen/` is generated. Never hand-edit.** Same rule the frontend applies to `BACKEND_HANDOFF.md`.
- **`internal/domain/` must never import `internal/gen/`.** Handlers translate between them.
- Money is `int64` minor units, always paired with a currency string. No floats, no `decimal`.
- Every tenant-owned table has `tenant_id uuid NOT NULL` and RLS enabled.
- Error responses use the contract's `Error` schema: `{"error":{"code":string,"message":string}}`.
- SQL strings inline in Go are acceptable only in `internal/store/` plumbing and tests. From Stage 1 onward, all domain queries live in `db/queries/*.sql` compiled by sqlc — sqlc is introduced in Stage 1, when the first real query exists, not here.
- Local ports: control-api `8080`, public-api `8081`, Postgres `5432`, Redis `6379`, ClickHouse HTTP `8123`.
- Local databases: `sms_dev`, `sms_test` (Postgres); `sms_dev`, `sms_test` (ClickHouse).

```bash
# Creates the local Postgres and ClickHouse databases, plus the unprivileged
# application role. The app role is deliberately NOT the table owner so that
# row-level security cannot be bypassed — see docs/ARCHITECTURE.md §7.
set -euo pipefail

for db in sms_dev sms_test; do
  psql -d postgres -tc "SELECT 1 FROM pg_database WHERE datname='$db'" | grep -q 1 \
    || psql -d postgres -c "CREATE DATABASE $db"

  || psql -d postgres -c "CREATE ROLE sms_app LOGIN PASSWORD 'sms_app_local'"

  clickhouse client --query "CREATE DATABASE IF NOT EXISTS $db"
done
```

```sql
CREATE EXTENSION IF NOT EXISTS pgcrypto;

CREATE TABLE tenants (
    id          uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    name        text        NOT NULL,
    status      text        NOT NULL DEFAULT 'active'
                            CHECK (status IN ('active','suspended','throttled')),
    created_at  timestamptz NOT NULL DEFAULT now(),
    updated_at  timestamptz NOT NULL DEFAULT now()
);

-- Every tenant-owned table follows this exact shape: RLS on, FORCE so even the
-- owner is subject to it, and a policy keyed to the per-transaction setting the
-- API sets from the verified token. See docs/ARCHITECTURE.md §7.

ALTER TABLE tenants FORCE ROW LEVEL SECURITY;

CREATE POLICY tenant_isolation ON tenants
    USING (id = current_setting('app.tenant_id', true)::uuid);

GRANT SELECT, INSERT, UPDATE ON tenants TO sms_app;

-- +goose Down
DROP TABLE tenants;
```

- [ ] **Step 5: Write the config loader**

`internal/platform/config/config.go`:

```go
// Package config loads process configuration from the environment. It fails
// fast on anything missing rather than starting with a half-configured process.
package config

import (
	"fmt"
	"os"
)

type Config struct {
	DatabaseURL      string
	DatabaseAdminURL string
	ClickHouseURL    string
	RedisURL         string
	ControlAPIAddr   string
	JWTSigningKey    string
	LogLevel         string
}

func Load() (Config, error) {
	cfg := Config{
		DatabaseURL:      os.Getenv("DATABASE_URL"),
		DatabaseAdminURL: os.Getenv("DATABASE_ADMIN_URL"),
		ClickHouseURL:    os.Getenv("CLICKHOUSE_URL"),
		RedisURL:         os.Getenv("REDIS_URL"),
		ControlAPIAddr:   envOr("CONTROL_API_ADDR", ":8080"),
		JWTSigningKey:    os.Getenv("JWT_SIGNING_KEY"),
		LogLevel:         envOr("LOG_LEVEL", "info"),
	}
	required := map[string]string{
		"DATABASE_URL":    cfg.DatabaseURL,
		"CLICKHOUSE_URL":  cfg.ClickHouseURL,
		"REDIS_URL":       cfg.RedisURL,
		"JWT_SIGNING_KEY": cfg.JWTSigningKey,
	}
	for name, value := range required {
		if value == "" {
			return Config{}, fmt.Errorf("config: %s is required", name)
		}
	}
	return cfg, nil
}

func envOr(name, fallback string) string {
	if v := os.Getenv(name); v != "" {
		return v
	}
	return fallback
}
```

```go
package store_test

import (
	"context"
	"os"
	"testing"

	"github.com/google/uuid"
	"github.com/jackc/pgx/v5"
	"github.com/saeedafri/sms-be/internal/store"
)

// Tenant A must never be able to read tenant B's row. This is enforced by
// Postgres row-level security, not by the query text — so the query here has
// deliberately NO tenant_id predicate. If RLS is misconfigured, this test
// finds a row and fails, which is exactly the defect we want caught.
func TestTenantCannotReadAnotherTenant(t *testing.T) {
	url := os.Getenv("TEST_DATABASE_URL")
	if url == "" {
		t.Skip("TEST_DATABASE_URL not set")
	}
	ctx := context.Background()

	pool, err := store.Open(ctx, url)
	if err != nil {
		t.Fatalf("Open: %v", err)
	}
	defer pool.Close()

	adminPool, err := store.Open(ctx, os.Getenv("TEST_DATABASE_ADMIN_URL"))
	if err != nil {
		t.Fatalf("Open admin: %v", err)
	}
	defer adminPool.Close()

	tenantA, tenantB := uuid.New(), uuid.New()
	for _, id := range []uuid.UUID{tenantA, tenantB} {
		_, err := adminPool.Exec(ctx,
			`INSERT INTO tenants (id, name) VALUES ($1, $2)`, id, "t-"+id.String())
		if err != nil {
			t.Fatalf("seed tenant: %v", err)
		}
	}
	t.Cleanup(func() {
		_, _ = adminPool.Exec(ctx, `DELETE FROM tenants WHERE id = ANY($1)`,
			[]uuid.UUID{tenantA, tenantB})
	})

	var found int
	err = store.WithTenant(ctx, pool, tenantA, func(tx pgx.Tx) error {
		return tx.QueryRow(ctx,
			`SELECT count(*) FROM tenants WHERE id = $1`, tenantB).Scan(&found)
	})
	if err != nil {
		t.Fatalf("WithTenant: %v", err)
	}
	if found != 0 {
		t.Fatalf("tenant A read %d of tenant B's rows; want 0 — RLS is not isolating", found)
	}
}
```

- [ ] **Step 4: Write the implementation**

`internal/store/pool.go`:

```go
// Package store owns database connectivity and the tenant-scoping contract.
// Every tenant-scoped read or write goes through WithTenant, which is what
// makes row-level security effective: the policy reads app.tenant_id, and
// this is the only place that setting is written.
package store

import (
	"context"
	"fmt"

	"github.com/google/uuid"
	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

func Open(ctx context.Context, url string) (*pgxpool.Pool, error) {
	pool, err := pgxpool.New(ctx, url)
	if err != nil {
		return nil, fmt.Errorf("store: open pool: %w", err)
	}
	if err := pool.Ping(ctx); err != nil {
		pool.Close()
		return nil, fmt.Errorf("store: ping: %w", err)
	}
	return pool, nil
}

// WithTenant runs fn inside a transaction scoped to one tenant. SET LOCAL ties
// the setting to this transaction, so it cannot leak to the next user of this
// pooled connection.
func WithTenant(ctx context.Context, pool *pgxpool.Pool, tenantID uuid.UUID,
	fn func(pgx.Tx) error) error {

	tx, err := pool.Begin(ctx)
	if err != nil {
		return fmt.Errorf("store: begin: %w", err)
	}
	defer func() { _ = tx.Rollback(ctx) }()

	if _, err := tx.Exec(ctx, `SELECT set_config('app.tenant_id', $1, true)`,
		tenantID.String()); err != nil {
		return fmt.Errorf("store: set tenant: %w", err)
	}
	if err := fn(tx); err != nil {
		return err
	}
	return tx.Commit(ctx)
}
```

**Why the Unimplemented base struct:** `oapi-codegen` emits one interface with all 151 methods, so nothing compiles until all 151 exist. Embedding a generated `Unimplemented` struct lets us implement operations stage by stage while keeping the compile-time guarantee for the ones we *have* implemented — if the spec changes a field type on an implemented operation, the build breaks. That is the property we're buying.

- [ ] **Step 4: Write the Unimplemented generator**

`cmd/gen-unimplemented/main.go`:

```go
// Generates internal/api/unimplemented.go: a struct implementing every method
// of the generated StrictServerInterface with a 501 response. Real handlers
// embed it and override the operations their stage owns, so the project
// compiles while only part of the contract is built.
package main

import (
	"bytes"
	"fmt"
	"go/ast"
	"go/parser"
	"go/printer"
	"go/token"
	"os"
)

// The generated interface's method signatures already carry every type they
// need, qualified with the gen package alias, so this generator only needs to
// copy them verbatim.

func main() {
	fset := token.NewFileSet()
	file, err := parser.ParseFile(fset, "internal/gen/api/control.gen.go", nil, 0)
	if err != nil {
		fatal(err)
	}

	var out bytes.Buffer
	out.WriteString("// Code generated by cmd/gen-unimplemented. DO NOT EDIT.\n\n")
	out.WriteString("package api\n\nimport (\n\t\"context\"\n\n")
	out.WriteString("\tgen \"github.com/saeedafri/sms-be/internal/gen/api\"\n)\n\n")
	out.WriteString("var _ = context.Background // the copied signatures use context\n\n")
	out.WriteString("// Unimplemented answers 501 for every contract operation.\n")
	out.WriteString("type Unimplemented struct{}\n\n")

	iface := findInterface(file, "StrictServerInterface")
	if iface == nil {
		fatal(fmt.Errorf("StrictServerInterface not found"))
	}
	for _, method := range iface.Methods.List {
		if len(method.Names) == 0 {
			continue
		}
		name := method.Names[0].Name
		fn := method.Type.(*ast.FuncType)
		params := render(fset, fn.Params)
		results := render(fset, fn.Results)
		fmt.Fprintf(&out,
			"func (Unimplemented) %s%s %s {\n\treturn nil, errNotImplemented(%q)\n}\n\n",
			name, params, results, name)
	}
	if err := os.WriteFile("internal/api/unimplemented.go", out.Bytes(), 0o644); err != nil {
		fatal(err)
	}
	fmt.Println("wrote internal/api/unimplemented.go")
}

func findInterface(file *ast.File, name string) *ast.InterfaceType {
	var found *ast.InterfaceType
	ast.Inspect(file, func(n ast.Node) bool {
		ts, ok := n.(*ast.TypeSpec)
		if !ok || ts.Name.Name != name {
			return true
		}
		if it, ok := ts.Type.(*ast.InterfaceType); ok {
			found = it
		}
		return false
	})
	return found
}

func render(fset *token.FileSet, node ast.Node) string {
	var buf bytes.Buffer
	_ = printer.Fprint(&buf, fset, node)
	return buf.String()
}

func fatal(err error) {
	fmt.Fprintln(os.Stderr, "gen-unimplemented:", err)
	os.Exit(1)
}
```

```go
package api

import (
	"encoding/json"
	"errors"
	"log/slog"
	"net/http"

	"github.com/go-chi/chi/v5"
	"github.com/go-chi/chi/v5/middleware"
	"github.com/jackc/pgx/v5/pgxpool"

	gen "github.com/saeedafri/sms-be/internal/gen/api"
)

// Server satisfies the generated contract interface. Embedding Unimplemented
// means every operation compiles today and answers 501; each stage replaces
// operations by defining real methods on Server, which shadow the embedded
// ones. Nothing else in the codebase changes when that happens.
type Server struct {
	Unimplemented
	DB     *pgxpool.Pool
	Logger *slog.Logger
}

func NewRouter(s *Server) http.Handler {
	r := chi.NewRouter()
	r.Use(middleware.RequestID, middleware.RealIP, middleware.Recoverer)

	r.Get("/healthz", func(w http.ResponseWriter, _ *http.Request) {
		writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
	})

	handler := gen.NewStrictHandlerWithOptions(s, nil, gen.StrictHTTPServerOptions{
		RequestErrorHandlerFunc:  func(w http.ResponseWriter, _ *http.Request, err error) {
			writeError(w, http.StatusBadRequest, "bad_request", err.Error())
		},
		ResponseErrorHandlerFunc: writeOperationError,
	})
	gen.HandlerFromMux(handler, r)
	return r
}

// Every error leaving an operation is rendered as the contract's Error schema.
// The frontend's error states read error.code, so the envelope is part of the
// contract even for failures we consider internal.
func writeOperationError(w http.ResponseWriter, _ *http.Request, err error) {
	var notImpl notImplementedError
	if errors.As(err, &notImpl) {
		writeError(w, http.StatusNotImplemented, "not_implemented", notImpl.Error())
		return
	}
	writeError(w, http.StatusInternalServerError, "internal_error", "an unexpected error occurred")
}

func writeError(w http.ResponseWriter, status int, code, message string) {
	writeJSON(w, status, map[string]any{
		"error": map[string]string{"code": code, "message": message},
	})
}

func writeJSON(w http.ResponseWriter, status int, body any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(body)
}
```

**Version note:** `NewStrictHandlerWithOptions`, `StrictHTTPServerOptions`, and its field names
(`RequestErrorHandlerFunc` / `ResponseErrorHandlerFunc`) come from the generated file. Before
writing this, open `internal/gen/api/control.gen.go` and grep for `func NewStrictHandler` and
`type StrictHTTPServerOptions` to confirm the exact names your oapi-codegen version emitted, and
use those. The generated file is the authority, not this plan.

```go
package main

import (
	"context"
	"errors"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"github.com/saeedafri/sms-be/internal/api"
	"github.com/saeedafri/sms-be/internal/platform/config"
	"github.com/saeedafri/sms-be/internal/platform/telemetry"
	"github.com/saeedafri/sms-be/internal/store"
)

func main() {
	cfg, err := config.Load()
	if err != nil {
		os.Stderr.WriteString(err.Error() + "\n")
		os.Exit(1)
	}
	logger := telemetry.NewLogger(cfg.LogLevel)

	ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
	defer stop()

	pool, err := store.Open(ctx, cfg.DatabaseURL)
	if err != nil {
		logger.Error("database unavailable", "error", err)
		os.Exit(1)
	}
	defer pool.Close()

	server := &http.Server{
		Addr:              cfg.ControlAPIAddr,
		Handler:           api.NewRouter(&api.Server{DB: pool, Logger: logger}),
		ReadHeaderTimeout: 10 * time.Second,
	}

	go func() {
		logger.Info("control-api listening", "addr", cfg.ControlAPIAddr)
		if err := server.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
			logger.Error("server failed", "error", err)
			os.Exit(1)
		}
	}()

	<-ctx.Done()
	logger.Info("shutting down")
	shutdownCtx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()
	_ = server.Shutdown(shutdownCtx)
}
```

```go
package store

import (
	"context"
	"fmt"
	"net/url"
	"strings"

	"github.com/ClickHouse/clickhouse-go/v2"
	"github.com/ClickHouse/clickhouse-go/v2/lib/driver"
)

// OpenClickHouse accepts an http://host:port/database URL for symmetry with
// the other connection strings, and dials ClickHouse's native protocol port,
// which is materially faster than HTTP for bulk inserts.
func OpenClickHouse(ctx context.Context, raw string) (driver.Conn, error) {
	parsed, err := url.Parse(raw)
	if err != nil {
		return nil, fmt.Errorf("store: parse clickhouse url: %w", err)
	}
	database := strings.TrimPrefix(parsed.Path, "/")
	if database == "" {
		database = "default"
	}
	conn, err := clickhouse.Open(&clickhouse.Options{
		Addr: []string{parsed.Hostname() + ":9000"},
		Auth: clickhouse.Auth{Database: database},
	})
	if err != nil {
		return nil, fmt.Errorf("store: open clickhouse: %w", err)
	}
	if err := conn.Ping(ctx); err != nil {
		conn.Close()
		return nil, fmt.Errorf("store: ping clickhouse: %w", err)
	}
	return conn, nil
}
```
